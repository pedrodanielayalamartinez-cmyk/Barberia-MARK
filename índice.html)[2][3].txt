    // Convierte "17:30" a "5:30 PM"
    function formatTime12h(time24) {
      const [h, m] = time24.split(':').map(Number);
      const period = h >= 12 ? 'PM' : 'AM';
      const hour12 = h % 12 === 0 ? 12 : h % 12;
      return `${hour12}:${String(m).padStart(2, '0')} ${period}`;
    }

    // Genera horarios exactos según el día de la semana (0=Dom, 1-5=Lun-Vie, 6=Sáb)
    function getScheduleForDate(dateStr) {
      const [y, m, d] = dateStr.split('-').map(Number);
      const day = new Date(y, m - 1, d).getDay();

      let startHour = 13;
      let endHour = 23; // Cierre 11:00 PM (último turno 10:30 PM)
      let label = 'Lunes a Viernes · 1:00 PM a 11:00 PM';
      let hasAfternoonBreak = false;

      if (day === 6) {
        startHour = 8;
        label = 'Sábado · 8:00 AM a 11:00 PM';
      } else if (day === 0) {
        startHour = 11;
        label = 'Domingo · 11:00 AM a 11:00 PM';
      } else {
        hasAfternoonBreak = true;
        label = 'Lunes a Viernes · 1:00 PM a 11:00 PM (Descanso 5:30 PM – 6:00 PM)';
      }

      const slots = [];
      for (let h = startHour; h < endHour; h++) {
        for (let min of [0, 30]) {
          const code = `${String(h).padStart(2, '0')}:${String(min).padStart(2, '0')}`;
          const isBreak = hasAfternoonBreak && code === '17:30';
          slots.push({ code, isBreak });
        }
      }

      return { label, slots };
    }

    }

    function compressImage(file, maxWidth = 650) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = (e) => {
          const img = new Image();
          img.onload = () => {
            const scale = Math.min(1, maxWidth / img.width);
            const canvas = document.createElement('canvas');
            canvas.width = img.width * scale;
            canvas.height = img.height * scale;
            canvas.getContext('2d').drawImage(img, 0, 0, canvas.width, canvas.height);
            resolve(canvas.toDataURL('image/jpeg', 0.78));
          };
          img.onerror = reject;
          img.src = e.target.result;
        };
        reader.onerror = reject;
        reader.readAsDataURL(file);
      });
    }

    function renderTimeSlots() {
      const selectedDate = dateInput.value;
      if (!selectedDate) return;

      const schedule = getScheduleForDate(selectedDate);
      document.getElementById('dayScheduleBanner').innerHTML = `
        <strong>Horario del día:</strong>
        <span>${escapeHTML(schedule.label)}</span>
      `;

      const appointments = getJSON(KEYS.APPTS, []);
      const bookedForDate = appointments
        .filter(a => a.date === selectedDate)
        .map(a => a.time);

      const now = new Date();
      const todayStr = getTodayString();
      const currentMinutes = now.getHours() * 60 + now.getMinutes();

      timeGrid.innerHTML = '';
      selectedTime = null;

      schedule.slots.forEach(({ code, isBreak }) => {
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.className = 'time-slot' + (isBreak ? ' break-slot' : '');

        const [h, m] = code.split(':').map(Number);
        const slotMinutes = h * 60 + m;
        const isPast = (selectedDate === todayStr && slotMinutes <= currentMinutes);
        const isBooked = bookedForDate.includes(code);

        if (isBreak) {
          btn.textContent = `${formatTime12h(code)} · Descanso`;
          btn.disabled = true;
        } else {
          btn.textContent = formatTime12h(code);
          if (isBooked || isPast) {
            btn.disabled = true;
          } else {
            btn.addEventListener('click', () => {
              document.querySelectorAll('.time-slot').forEach(b => b.classList.remove('selected'));
              btn.classList.add('selected');
              selectedTime = code;
              updateSummary();
            });
          }
        }

        timeGrid.appendChild(btn);
      });

      updateSummary();
    }

      const duration = selected.reduce((sum, c) => sum + Number(c.dataset.duration), 0);

      document.getElementById('sumDuration').textContent = `${duration} min`;
      document.getElementById('sumDate').textContent = dateInput.value || '-';
      document.getElementById('sumTime').textContent = selectedTime ? formatTime12h(selectedTime) : 'Por elegir';
      document.getElementById('sumTotal').textContent = `$${total} MXN`;
      document.getElementById('formFeedback').textContent = '';
    }

    document.getElementById('bookingForm').addEventListener('submit', async (e) => {
      e.preventDefault();
      const session = getSession();
      const feedback = document.getElementById('formFeedback');
      const confirmBtn = document.getElementById('confirmBtn');

      if (!session || session.role !== 'client') {
        feedback.textContent = 'Debes iniciar sesión para reservar.';
        return;
      }

      const selectedServices = Array.from(checkboxes).filter(c => c.checked);
      if (!selectedServices.length) {
        feedback.textContent = 'Selecciona al menos un servicio.';
        return;
      }

      if (!selectedTime) {
        feedback.textContent = 'Elige un horario disponible.';
        return;
      }

      const serviceNames = selectedServices.map(c => c.dataset.name);
      const total = selectedServices.reduce((sum, c) => sum + Number(c.dataset.price), 0);
      const date = dateInput.value;
      const note = document.getElementById('clientNote').value.trim();
      const folio = 'MARK-' + Math.floor(1000 + Math.random() * 9000);

      confirmBtn.disabled = true;
      confirmBtn.textContent = 'Enviando confirmación...';

      const newAppt = {
        id: 'c_' + Date.now(),
        folio,
        userId: session.userId,
        clientName: session.name,
        clientPhone: session.phone || 'No registrado',
        services: serviceNames,
        total,
        date,
        time: selectedTime,
        note,
        status: 'Pendiente',
        photo: null,
        barberComment: '',
        serverConfirmationMsg: ''
      };

      const dispatchResult = await dispatchServerConfirmation(newAppt);
      newAppt.serverConfirmationMsg = dispatchResult.message;

      const appts = getJSON(KEYS.APPTS, []);
      appts.push(newAppt);
      setJSON(KEYS.APPTS, appts);

      confirmBtn.disabled = false;
      confirmBtn.textContent = 'Confirmar cita';

      document.getElementById('modalDetails').textContent =
        `Folio ${folio} · ${date} a las ${formatTime12h(selectedTime)}`;

      document.getElementById('serverDispatchBox').innerHTML = `
        <strong>✓ Mensaje de confirmación enviado por el servidor (${escapeHTML(newAppt.clientPhone)}):</strong>
        <span>"${escapeHTML(dispatchResult.message)}"</span>
      `;

      document.getElementById('confirmModal').classList.add('active');

      e.target.reset();
      dateInput.value = date;
      renderTimeSlots();
      renderClientRecord();
    });

    function renderClientRecord() {
      const session = getSession();
      const container = document.getElementById('clientRecordList');
      if (!session || session.role !== 'client') return;

      const myAppts = getJSON(KEYS.APPTS, [])
        .filter(a => a.userId === session.userId)
        .sort((a, b) => (b.date + b.time).localeCompare(a.date + a.time));

      if (!myAppts.length) {
        container.innerHTML = `<p style="color:var(--text-muted); font-size:0.9rem;">Aún no tienes citas registradas.</p>`;
        return;
      }

      container.innerHTML = myAppts.map(a => {
        const pillClass = a.status === 'Completada' ? 'status-completada' : 'status-pendiente';
        return `
          <div class="record-card">
            <div>
              <div style="display:flex; justify-content:space-between; align-items:center; margin-bottom:10px;">
                <strong style="font-size:0.95rem;">${escapeHTML(a.date)} · ${escapeHTML(formatTime12h(a.time))}</strong>
                <span class="status-pill ${pillClass}">${escapeHTML(a.status)}</span>
              </div>
              <p style="font-size:0.9rem; color:var(--text-secondary); margin-bottom:4px;">
                ${escapeHTML(a.services.join(', '))}
              </p>
              <strong style="font-size:0.92rem;">$${a.total} MXN</strong>
              ${a.serverConfirmationMsg ? `<div class="server-msg-tag">✓ Confirmación enviada (${escapeHTML(a.folio || 'MARK')})</div>` : ''}
              ${a.barberComment ? `<p style="font-size:0.82rem; color:var(--text-muted); margin-top:8px;">Nota barbero: ${escapeHTML(a.barberComment)}</p>` : ''}
            </div>

            ${a.photo ? `
              <div>
                <img src="${a.photo}" alt="Foto de corte" class="haircut-photo" style="cursor:pointer;"
                     onclick="openLightbox('${a.id}')">
                <span style="display:block; font-size:0.78rem; color:var(--cyan-frost); margin-top:6px;">Foto subida por Barbería Mark</span>
              </div>
            ` : `
              <div style="display:flex; justify-content:space-between; align-items:center;">
                <span style="font-size:0.8rem; color:var(--text-muted);">
                  ${a.status === 'Pendiente' ? 'En espera de atención' : 'Sin foto adjunta'}
                </span>
                ${a.status === 'Pendiente' ? `<button type="button" class="btn-secondary btn-sm" onclick="cancelMyAppt('${a.id}')">Cancelar</button>` : ''}
              </div>
            `}
          </div>
        `;
      }).join('');
    }

    function cancelMyAppt(id) {
      const appts = getJSON(KEYS.APPTS, []).filter(a => a.id !== id);
      setJSON(KEYS.APPTS, appts);
      renderTimeSlots();
      renderClientRecord();
    }

    function setAdminTab(tab, btn) {

    function renderAdminDashboard() {
      const appts = getJSON(KEYS.APPTS, []);
      const users = getJSON(KEYS.USERS, []);

      const pending = appts.filter(a => a.status === 'Pendiente');
      const done = appts.filter(a => a.status === 'Completada');
      const revenue = appts.reduce((sum, a) => sum + Number(a.total || 0), 0);

      document.getElementById('kpiPending').textContent = pending.length;
      document.getElementById('kpiDone').textContent = done.length;
      document.getElementById('kpiClients').textContent = users.length;
      document.getElementById('kpiRevenue').textContent = `$${revenue}`;

      const area = document.getElementById('adminContentArea');

      if (adminTab === 'Clientes') {
        if (!users.length) {
          area.innerHTML = `<p style="color:var(--text-muted);">Aún no hay clientes registrados.</p>`;
          return;
        }
        area.innerHTML = `<div class="clients-grid">` + users.map(u => {
          const clientAppts = appts.filter(a => a.userId === u.id);
          const photos = clientAppts.filter(a => a.photo);
          return `
            <div class="client-profile-card">
              <div style="display:flex; justify-content:space-between; align-items:center;">
                <h4>${escapeHTML(u.name)}</h4>
                <span class="status-pill status-pendiente">${clientAppts.length} cita(s)</span>
              </div>
              <p style="font-size:0.84rem; color:var(--text-secondary); margin-top:4px;">WhatsApp: ${escapeHTML(u.phone || '-')}</p>
              <p style="font-size:0.8rem; color:var(--text-muted);">Registrado: ${escapeHTML(u.createdAt || '-')}</p>
              ${photos.length ? `
                <div class="client-photos-strip">
                  ${photos.map(p => `<img src="${p.photo}" alt="Corte" onclick="openLightbox('${p.id}')">`).join('')}
                </div>
              ` : `<p style="font-size:0.82rem; color:var(--text-muted); margin-top:14px;">Sin fotos de corte todavía.</p>`}
            </div>
          `;
        }).join('') + `</div>`;
        return;
      }

      let filtered = appts;
      if (adminTab === 'Pendiente') filtered = pending;
      if (adminTab === 'Completada') filtered = done;

      filtered = [...filtered].sort((a, b) => (a.date + a.time).localeCompare(b.date + b.time));

      if (!filtered.length) {
        area.innerHTML = `
          <div class="kpi-card" style="text-align:center; color:var(--text-muted);">
            No hay citas en la categoría "${escapeHTML(adminTab)}".
          </div>`;
        return;
      }

      area.innerHTML = `<div class="admin-list">` + filtered.map(a => {
        const pillClass = a.status === 'Completada' ? 'status-completada' : 'status-pendiente';
        return `
          <div class="admin-appt-card">
            <div class="admin-datetime">
              <strong>${escapeHTML(formatTime12h(a.time))}</strong>
              <span>${escapeHTML(a.date)}</span>
            </div>

            <div class="admin-client-info">
              <h4>
                ${escapeHTML(a.clientName)}
                <span class="status-pill ${pillClass}">${escapeHTML(a.status)}</span>
              </h4>
              <p>${escapeHTML(a.services.join(', '))} · <strong>$${a.total} MXN</strong> · Tel: ${escapeHTML(a.clientPhone || '-')}</p>
              ${a.note ? `<p style="font-size:0.82rem; color:var(--text-muted);">Nota cliente: ${escapeHTML(a.note)}</p>` : ''}
              ${a.barberComment ? `<p style="font-size:0.82rem; color:var(--cyan-frost);">Registro: ${escapeHTML(a.barberComment)}</p>` : ''}
            </div>

            <div class="admin-actions">
              ${a.photo ? `<img src="${a.photo}" class="thumb-mini" title="Ver foto" onclick="openLightbox('${a.id}')">` : ''}
              <button type="button" class="btn-primary btn-sm" onclick="openPhotoModal('${a.id}')">
                ${a.status === 'Pendiente' ? 'Finalizar / Subir foto' : (a.photo ? 'Cambiar foto' : 'Subir foto')}
              </button>
              <button type="button" class="btn-secondary btn-sm" onclick="deleteApptAdmin('${a.id}')">Eliminar</button>
            </div>
          </div>
        `;
      }).join('') + `</div>`;
    }

    // ================= SUBIDA DE FOTOS DEL CORTE =================
    function openPhotoModal(apptId) {
      const appts = getJSON(KEYS.APPTS, []);
      const appt = appts.find(a => a.id === apptId);
      if (!appt) return;

      currentEditingApptId = apptId;
      currentPhotoBase64 = appt.photo || null;

      document.getElementById('photoModalTitle').textContent =
        appt.status === 'Pendiente' ? 'Finalizar corte y subir foto' : 'Actualizar foto del corte';
      document.getElementById('photoModalSub').textContent =
        `Cliente: ${appt.clientName} · ${appt.services.join(', ')} (${appt.date})`;
      document.getElementById('barberComment').value = appt.barberComment || '';
      document.getElementById('photoFileInput').value = '';

      const preview = document.getElementById('photoPreviewBox');
      preview.innerHTML = currentPhotoBase64
        ? `<img src="${currentPhotoBase64}" alt="Vista previa">`
        : `<span>Sin foto seleccionada (opcional)</span>`;

      document.getElementById('photoModal').classList.add('active');
    }

    document.getElementById('photoFileInput').addEventListener('change', async (e) => {
      const file = e.target.files[0];
      if (!file) return;
      try {
        currentPhotoBase64 = await compressImage(file);
        document.getElementById('photoPreviewBox').innerHTML = `<img src="${currentPhotoBase64}" alt="Vista previa">`;
      } catch {
        alert('No se pudo procesar la imagen.');
      }
    });

    function saveCompletedAppointment() {
      if (!currentEditingApptId) return;
      const appts = getJSON(KEYS.APPTS, []);
      const index = appts.findIndex(a => a.id === currentEditingApptId);
      if (index === -1) return;

      appts[index].status = 'Completada';
      appts[index].photo = currentPhotoBase64;
      appts[index].barberComment = document.getElementById('barberComment').value.trim();

      if (setJSON(KEYS.APPTS, appts)) {
        closeModal('photoModal');
        renderAdminDashboard();
      }
    }

    function openLightbox(apptId) {
      const appt = getJSON(KEYS.APPTS, []).find(a => a.id === apptId);
      if (!appt || !appt.photo) return;
      document.getElementById('lightboxImg').src = appt.photo;
      document.getElementById('lightboxCaption').textContent =
        `${appt.clientName} · ${appt.services.join(', ')} (${appt.date})${appt.barberComment ? ' — ' + appt.barberComment : ''}`;
      document.getElementById('lightboxModal').classList.add('active');
    }

    function closeModal(id) {
      document.getElementById(id).classList.remove('active');
